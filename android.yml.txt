
name: Build Pak Scanner APK

on:
  workflow_dispatch:
  push:
    branches:
      - main
      - master

permissions:
  contents: read

jobs:
  build:
    runs-on: ubuntu-latest

    steps:
      - name: Checkout repository
        uses: actions/checkout@v4

      - name: Extract ZIP and find project
        shell: bash
        run: |
          set -e
          ZIP=$(find . -type f -name '*.zip' -not -path '*/.git/*' | head -n 1)

          if [ -n "$ZIP" ]; then
            mkdir -p extracted
            unzip -oq "$ZIP" -d extracted
          fi

          PROJECT=$(find "$PWD" -type f \( -name settings.gradle -o -name settings.gradle.kts \) -not -path '*/.git/*' -not -path '*/.gradle/*' | head -n 1)

          if [ -z "$PROJECT" ]; then
            echo "ERROR: settings.gradle not found."
            exit 1
          fi

          echo "PROJECT_DIR=$(dirname "$PROJECT")" >> "$GITHUB_ENV"

      - name: Set up Java
        uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '17'

      - name: Set up Android SDK
        uses: android-actions/setup-android@v4

      - name: Verify Android SDK
        shell: bash
        run: |
          set -e
          echo "ANDROID_HOME=$ANDROID_HOME"
          echo "ANDROID_SDK_ROOT=$ANDROID_SDK_ROOT"
          test -d "$ANDROID_HOME"
          sdkmanager --version
          set +o pipefail
          yes | sdkmanager --licenses || true
          set -o pipefail

      - name: Install Gradle
        uses: gradle/actions/setup-gradle@v4
        with:
          gradle-version: '8.14.3'

      - name: Generate Gradle Wrapper
        working-directory: ${{ env.PROJECT_DIR }}
        shell: bash
        run: |
          set -e
          if [ ! -f gradlew ]; then
            gradle wrapper --gradle-version 8.14.3
          fi
          chmod +x gradlew

      - name: Fix Java and Kotlin JVM targets
        working-directory: ${{ env.PROJECT_DIR }}
        shell: bash
        run: |
          set -e
          python3 <<'PY'
          from pathlib import Path
          import re

          p = Path("app/build.gradle")
          if not p.exists():
              p = Path("app/build.gradle.kts")

          if not p.exists():
              raise SystemExit("ERROR: app/build.gradle or app/build.gradle.kts not found")

          s = p.read_text()

          if p.name == "build.gradle":
              if re.search(r"compileOptions\s*\{", s):
                  s = re.sub(r"sourceCompatibility\s+JavaVersion\.VERSION_1_8", "sourceCompatibility JavaVersion.VERSION_17", s)
                  s = re.sub(r"targetCompatibility\s+JavaVersion\.VERSION_1_8", "targetCompatibility JavaVersion.VERSION_17", s)
              else:
                  s = s.replace("android {", "android {\n    compileOptions {\n        sourceCompatibility JavaVersion.VERSION_17\n        targetCompatibility JavaVersion.VERSION_17\n    }", 1)

              if re.search(r"kotlinOptions\s*\{", s):
                  s = re.sub(r"kotlinOptions\s*\{[^}]*\}", 'kotlinOptions {\n        jvmTarget = "17"\n    }', s, count=1)
              else:
                  s = s.replace("android {", 'android {\n    kotlinOptions {\n        jvmTarget = "17"\n    }', 1)
          else:
              if re.search(r"compileOptions\s*\{", s):
                  s = re.sub(r"sourceCompatibility\s*=\s*JavaVersion\.VERSION_1_8", "sourceCompatibility = JavaVersion.VERSION_17", s)
                  s = re.sub(r"targetCompatibility\s*=\s*JavaVersion\.VERSION_1_8", "targetCompatibility = JavaVersion.VERSION_17", s)
              else:
                  s = s.replace("android {", "android {\n    compileOptions {\n        sourceCompatibility = JavaVersion.VERSION_17\n        targetCompatibility = JavaVersion.VERSION_17\n    }", 1)

              if re.search(r"kotlinOptions\s*\{", s):
                  s = re.sub(r"kotlinOptions\s*\{[^}]*\}", 'kotlinOptions {\n        jvmTarget = "17"\n    }', s, count=1)
              else:
                  s = s.replace("android {", 'android {\n    kotlinOptions {\n        jvmTarget = "17"\n    }', 1)

          p.write_text(s)
          print(f"Updated JVM targets in {p}")
          PY

      - name: Build debug APK
        working-directory: ${{ env.PROJECT_DIR }}
        run: ./gradlew --no-daemon assembleDebug

      - name: Upload APK
        uses: actions/upload-artifact@v4
        with:
          name: Pak-Scanner-APK
          path: ${{ env.PROJECT_DIR }}/app/build/outputs/apk/debug/*.apk
          if-no-files-found: error
